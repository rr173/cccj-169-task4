离线核对（命令行）

        
# 在没有服务进程、没有数据库的机器上：
# 1) 只读核对（设备/版本/事件数/完整性证明）
python3 -m app.sealctl verify 封包.json \
    --device DEV-XXXX --secret-file device.key
# 2) 导入本地操作员台账（拒绝重复导入与失效版本）
python3 -m app.sealctl import 封包.json \
    --device DEV-XXXX --secret-file device.key
# 3) 只看头部（不验签）
python3 -m app.sealctl show 封包.json

# 拒绝码（退出码）：
#   seal_tampered(3)         封存包被改动
#   seal_cross_device(4)     换设备使用
#   seal_duplicate_import(5) 重复导入
#   seal_version_stale(6)    引用已失效版本
